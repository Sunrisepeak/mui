namespace mlibs::memory {
  class Arena {
    explicit Arena(std::size_t block_size = 64 * 1024) : block_size_(block_size);
    Arena(const Arena&) = delete;
    Arena& operator=(const Arena&) = delete;
    Arena(Arena&&) noexcept = default;
    Arena& operator=(Arena&&) noexcept = default;
    [[nodiscard]] void* allocate(std::size_t size, std::size_t align);
    template <class T, class... Args> requires std::is_trivially_destructible_v<T> [[nodiscard]] T* make(Args&&... args);
    template <class T> requires std::is_trivially_destructible_v<T> && std::is_trivially_default_constructible_v<T> [[nodiscard]] std::span<T> array(std::size_t n);
    template <class T> requires std::is_trivially_destructible_v<T> && std::is_trivially_copyable_v<T> [[nodiscard]] std::span<T> copy(std::span<const T> from);
    [[nodiscard]] std::string_view copy(std::string_view text);
    void reset() noexcept;
    [[nodiscard]] std::size_t used() const noexcept;
    [[nodiscard]] std::size_t capacity() const noexcept;
    [[nodiscard]] std::size_t heap_allocations() const noexcept;
  };
}

== mlibs.memory.api:handle
namespace mlibs::memory {
  template <class Tag> struct Handle {
    static constexpr std::uint32_t invalid_index = std::numeric_limits<std::uint32_t>::max();
    std::uint32_t index = invalid_index;
    std::uint32_t generation = 0;
    [[nodiscard]] constexpr bool valid() const noexcept;
    constexpr explicit operator bool() const noexcept;
    [[nodiscard]] constexpr std::uint64_t bits() const noexcept;
    [[nodiscard]] static constexpr Handle from_bits(std::uint64_t b) noexcept;
    friend constexpr bool operator==(Handle a, Handle b) noexcept;
  };
  template <class Tag> struct OwnedHandle {
    static constexpr std::uint32_t invalid_slot = std::numeric_limits<std::uint32_t>::max();
    std::uint64_t owner = 0;
    std::uint32_t slot = invalid_slot;
    std::uint64_t generation = 0;
    [[nodiscard]] constexpr explicit operator bool() const noexcept;
    friend constexpr bool operator==(const OwnedHandle&, const OwnedHandle&) noexcept = default;
  };
  template <class T, class Tag> class SlotMap {
    using handle_type = Handle<Tag>;
    handle_type insert(T value);
    bool erase(handle_type key);
    [[nodiscard]] bool contains(handle_type key) const noexcept;
    [[nodiscard]] T* get(handle_type key) noexcept;
    [[nodiscard]] const T* get(handle_type key) const noexcept;
    [[nodiscard]] std::size_t size() const noexcept;
    [[nodiscard]] bool empty() const noexcept;
    template <class F> void for_each(F&& function);
    template <class F> void for_each(F&& function) const;
    void clear();
  };
}
