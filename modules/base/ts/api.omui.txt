  struct NamedModule {
    std::string name;
    std::string source;
  };
  struct DataOptions {
    std::filesystem::path root;
    std::size_t memory_bytes = 64 * 1024 * 1024;
    std::chrono::milliseconds time{…};
    std::vector<NamedModule> modules;
  };
  struct Data {
    std::string kind;
    std::string json;
    std::string format;

  };
  struct DefinedKind {
    std::string_view name;
    int major = 1;
  };
