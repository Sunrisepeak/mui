namespace oweb {
  struct Heading {
    int level = 1;
    std::string text;
    std::string anchor;
  };
  struct Route {
    std::string path;
    std::string source;
    std::string title;
    std::string language;
    std::vector<Heading> headings;
  };
  struct SiteOptions {
    std::string title;
    std::vector<std::string> languages{…};
    std::string url;
  };
  struct Page {
    Route route;
    std::string html;
    std::string markdown;
    std::size_t nodes = 0;
  };
  struct File {
    std::string path;
    std::uintmax_t bytes = 0;
  };
  struct Built {
    std::vector<Route> routes;
    std::vector<File> files;
  };
