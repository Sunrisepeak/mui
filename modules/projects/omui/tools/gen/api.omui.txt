namespace omui::gen {
  inline std::string utf8(const std::filesystem::path& p);
  inline std::string generic_utf8(const std::filesystem::path& p);
  struct Loc {
    std::string file;
    int line = 0;
    int column = 0;
  };
  struct Diagnostic {
    enum class Severity : std::uint8_t { error, warning };
    Severity severity = Severity::error;
    Loc loc;
    std::string message;
    std::string hint;
  };
  struct Named {
    std::string name;
    std::string doc;
    Loc loc;
  };
  struct Arg {
    std::string name;
    std::string type;
  };

    std::string aria;
    std::string accesskit;
    bool interactive = false;
  };
  struct State : Named {
  };
  struct Effect : Named {
    int tier = 0;
    bool answer = false;
  };
  struct Origin : Named {
  };
  struct Action : Named {
    std::string effect;
    std::vector<Arg> args;
  };
  struct Extension : Named {
    int version = 1;
    std::vector<std::string> requires_;
  };
  struct Profile : Named {
    std::vector<std::string> scripts;
    int min_scripts = 0;
  };
  struct MotionToken : Named {
    std::optional<double> duration_ms;
    std::string easing;
    std::optional<double> stiffness, damping, mass;
  };
  struct MotionSlot : Named {
  };
  struct MotionRecipe : Named {
    std::vector<std::string> primitives;
    std::string timing;
    std::string effect;
  };
  struct MotionPresence : Named {
    std::string kind;
    std::vector<std::string> options;
  };
  struct ExprOperator : Named {
    std::string text;
    int arity = 0;
    std::map<std::string, std::string> combinators;
  };
  struct ExprFunction : Named {
    std::string signature;
    std::map<std::string, std::string> combinators;
  };
  struct AbiField {
    std::string name;
    std::string type;
    std::string doc;
    int array = 0;
  };
  struct AbiParam {
    std::string name;
    std::string type;
  };
  struct AbiHandle : Named {
    std::string kind;
  };
  struct AbiResult : Named {
    int value = 0;
  };
  struct AbiEnum : Named {
    std::vector<std::string> values;
  };
  struct AbiStruct : Named {
    bool extensible = false;
    std::vector<AbiField> fields;
  };
  struct AbiCallback : Named {
    std::string returns;
    std::vector<AbiParam> params;
  };
  struct AbiFunction : Named {
    std::vector<AbiParam> params;
  };
  struct Abi {

    std::vector<AbiEnum> enums;
    std::vector<AbiStruct> structs;
    std::vector<AbiCallback> callbacks;
    std::vector<AbiFunction> functions;
  };
  using Default = std::variant<std::monostate, bool, std::int64_t, double, std::string, std::vector<std::string>>;
  struct Prop : Named {
    std::string type;
    std::vector<std::string> values;
    Default default_value;
    bool bindable = false;
  };
  struct Event : Named {
    std::vector<Arg> args;
    std::string coalesce;
  };
  struct ContractAction : Named {
    std::string effect;
    std::vector<Arg> args;
    std::string answer;
  };
  struct Key {
    std::string chord;
    std::string target;
    Loc loc;
  };

  };

  };
