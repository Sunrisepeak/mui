== mrender.effect
export import mrender.effect.api;
== mrender.effect.api
namespace mrender::effect {
  struct Diagnostic {
    std::string file;
    int line = 0;
    int column = 0;
    std::string message;
    std::string hint;
    std::string at;
    [[nodiscard]] std::string text() const;
  };
  using Diagnostics = std::vector<Diagnostic>;
  struct Param {
    std::string name;
    float value = 0;
    float min = 0, max = 1;
    std::string doc;
  };
  struct Declaration {
    std::string name;
    std::string doc;
    std::filesystem::path shader;
    std::string entry;
    std::vector<Param> params;
    float reach = 0;
    std::string reach_param;
  };
  [[nodiscard]] std::expected<Declaration, Diagnostics> read(std::string_view text, std::string_view file);
  struct Options {
    mlibs::rhi::ShaderFormats formats = mlibs::rhi::formats({…});
    std::filesystem::path slangc;
    std::function<std::expected<std::string, std::string>(const std::filesystem::path&)> data;
  };
  struct Loaded {
    std::filesystem::path file;
    std::string name;
    mrender::EffectRef ref;
    std::chrono::microseconds took{…};
  };
  struct Reload {
    std::filesystem::path file;
    std::string name;
    mrender::EffectRef ref;
    std::chrono::microseconds took{…};
    Diagnostics problems;
    [[nodiscard]] bool ok() const noexcept;
  };
  class Library {
    explicit Library(Options options = {…});
    Library(Library&&) noexcept;
    Library& operator=(Library&&) noexcept;
    ~Library();
    [[nodiscard]] std::expected<Loaded, Diagnostics> load(const std::filesystem::path& file);
    [[nodiscard]] std::expected<std::vector<Loaded>, Diagnostics> load_directory(const std::filesystem::path& dir);
    [[nodiscard]] std::vector<Reload> poll();
    [[nodiscard]] std::vector<std::string> names() const;
  };
}
