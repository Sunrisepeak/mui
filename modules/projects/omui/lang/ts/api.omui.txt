    std::string name, type, doc;
  };

    std::string name;
    std::vector<FieldShape> args;
  };
  struct ComponentShape {
    std::string name, doc;
    std::vector<PropShape> props;
    std::vector<EventShape> events;
    std::string origin;
    bool spec = false;
  };

    std::string name, doc;
