  struct PictureLayer {
    std::span<const std::uint8_t> rgba;
    int width = 0, height = 0;
    std::span<const std::uint32_t> objects;
    std::function<void(int x, int y, std::string& members)> describe;
  };
  struct FrameLayer {
    std::string_view name;
    const DisplayList* list = nullptr;
    Resources resources;
    float scale = 1.0f;
    std::optional<Affine> transform;
    std::optional<float> effect_scale;
    bool object_ids = false;
    const PictureLayer* picture = nullptr;
  };
  struct FrameOptions {
    Color background = Color::transparent();
    bool clear = true;
    std::span<const RowSpan> rows;
    Parallel* parallel = nullptr;
    std::int32_t rows_per_task = 64;
    bool measure = false;
    std::function<std::optional<xrender::raster::ImageView>(std::uint64_t id)> external;
  };
  struct LayerReport {
    std::string name;
    bool object_ids = false;
    RenderReport report;
  };
  struct FrameReport {
    int width = 0, height = 0;
    std::vector<LayerReport> layers;
  };
  struct PixelSource {
    std::int32_t layer = -1;
    std::uint32_t object = 0;
  };
  class CpuFrame {
    const FrameReport& render(std::span<const FrameLayer> layers, raster::Surface& target, const FrameOptions& options = {…});
    [[nodiscard]] const FrameReport& report() const noexcept;
    [[nodiscard]] PixelSource at(int x, int y) const noexcept;
    [[nodiscard]] std::span<const std::uint32_t> object_ids(std::size_t layer) const noexcept;
    void describe(std::size_t layer, int x, int y, std::string& members) const;
  };
  struct GpuFrameOptions {
    Color background = Color::transparent();
    bool clear = true;
    std::function<std::optional<mlibs::rhi::Texture>(std::uint64_t id)> external;
    std::size_t glyph_masks = std::numeric_limits<std::size_t>::max();
  };
  class GpuFrame {
    [[nodiscard]] static mlibs::rhi::Result<GpuFrame> create(mlibs::rhi::Device& device);
    [[nodiscard]] mlibs::rhi::Result<const FrameReport*> render(std::span<const FrameLayer> layers, mlibs::rhi::Texture target, std::uint32_t width, std::uint32_t height, const GpuFrameOptions& options = {…});
    [[nodiscard]] const FrameReport& report() const noexcept;
    [[nodiscard]] mlibs::rhi::Device& device() const noexcept;
  };
  class FrameSubject : public mlibs::agent::Subject {
    void show(const CpuFrame& frame, const raster::Surface& target, std::function<void()> before = {…});

    [[nodiscard]] std::string_view interface_name() const noexcept override;
    [[nodiscard]] mlibs::agent::Capabilities capabilities() const noexcept override;
    [[nodiscard]] std::span<const mlibs::control::CatalogEntry> operations() const noexcept override;
    [[nodiscard]] std::string_view listing() const noexcept override;
    [[nodiscard]] mlibs::control::Guide guide() const noexcept override;
    [[nodiscard]] std::uint64_t state_hash() const override;
    void operate(std::string_view operation, const mlibs::agent::Request& request, mlibs::agent::Answer& answer) override;
  };
