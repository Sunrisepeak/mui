== omui.player
namespace omui::player {
  struct Script {
    std::string lang;
    std::filesystem::path entry;
  };
  struct Manifest {
    std::string id = "openmui-app";
    std::string title = "openmui";
    Size size{…};
    std::filesystem::path base;
    std::vector<std::filesystem::path> components;
    std::vector<std::filesystem::path> views;
    std::vector<std::filesystem::path> themes;
    std::vector<std::string> theme;
    std::filesystem::path effects;
    std::vector<Script> scripts;
  };
  using Status = std::expected<void, Failure>;
  class Engine {
    virtual ~Engine() = default;
    [[nodiscard]] virtual std::future<Status> run(std::string code, std::string name) = 0;
    [[nodiscard]] virtual std::future<Status> reload(std::string code, std::string name) = 0;
    [[nodiscard]] virtual Status reload_now(std::string code, std::string name);
    virtual void module_changed(const std::filesystem::path& file, std::optional<std::string> text);
    virtual void keep_built_at(bool on);
    [[nodiscard]] virtual std::vector<std::filesystem::path> modules() const;
    [[nodiscard]] virtual bool idle() const = 0;
  };
  struct SourceEditor {
    std::function<std::expected<std::string, Failure>(std::string_view text, std::size_t line, std::size_t column, std::string_view property, std::optional<std::string_view> value)> set;
    std::function<std::expected<std::string, Failure>(std::string_view text, std::size_t line, std::size_t column, std::string_view element, std::optional<std::size_t> index)> insert;
    std::function<std::expected<std::string, Failure>(std::string_view text, std::size_t line, std::size_t column)> remove;
  };
  struct Language {
    std::string name;
    std::vector<std::string> extensions;
    std::function<std::unique_ptr<Engine>(abi::OmuiRuntime)> make;
    std::function<std::expected<std::string, Failure>(const std::filesystem::path&)> load;
    std::function<std::expected<std::string, Failure>(std::string_view text, const std::filesystem::path& file)> load_text;
    std::function<std::string(std::string_view catalog_json)> describe;
    std::function<std::expected<std::vector<std::filesystem::path>, Failure>( std::string_view catalog_json, std::string_view natives_json, const std::filesystem::path& root, const std::filesystem::path& dir)> declare;
    SourceEditor edit;
    std::function<std::expected<std::string, Failure>(const std::filesystem::path&)> data;
  };
  class Player;
  class Twin {
    virtual ~Twin() = default;
    [[nodiscard]] virtual Player& player() = 0;
    [[nodiscard]] virtual Runtime& runtime() = 0;
    [[nodiscard]] virtual seconds time() = 0;
    virtual void step(std::optional<seconds> at = std::nullopt) = 0;
    virtual void settle() = 0;
    [[nodiscard]] virtual std::expected<std::vector<std::uint8_t>, ViewFailure> png() = 0;
    [[nodiscard]] virtual std::expected<std::vector<std::uint8_t>, ViewFailure> framed_png(Rect frame, std::string_view fixed) = 0;
  };
  using TwinFn = std::function<std::expected<std::unique_ptr<Twin>, Failure>(bool design)>;
  class Designer {
    virtual ~Designer() = default;
  };
  using DesignerFactory = std::unique_ptr<Designer> (*)(Player& player);
  void provide_designer(DesignerFactory factory) noexcept;
  struct PlayerOptions {
    bool design = false;
    TwinFn twin;
  };
  class Player {
    Player(Runtime& runtime, Manifest manifest, std::vector<Language> languages, PlayerOptions options = {…});
    ~Player();
    Player(const Player&) = delete;
    Player& operator=(const Player&) = delete;
    Status start();
    std::vector<std::string> poll();
    Status show(const std::map<std::string, std::string>& texts);
    struct Source {
      std::string path;
      std::string language;
      std::string imported_by;
      bool edited = false;
    };
    [[nodiscard]] std::vector<Source> sources() const;
    [[nodiscard]] std::optional<std::string> text(std::string_view path) const;
    std::expected<Value, Failure> edit(std::string_view path, std::string_view text);
    std::expected<Value, Failure> commit_edits();
    Value discard_edits();
    [[nodiscard]] const Language* language_of(std::string_view path) const;
    [[nodiscard]] std::string source_name(const std::filesystem::path& file) const;
    [[nodiscard]] std::filesystem::path source_file(std::string_view path) const;
    [[nodiscard]] Runtime& runtime() const noexcept;
    [[nodiscard]] const PlayerOptions& options() const noexcept;
    [[nodiscard]] const std::vector<Language>& languages() const noexcept;
    void watch(seconds period = seconds(0.1));
    [[nodiscard]] bool idle() const;
    [[nodiscard]] std::span<const std::string> problems() const noexcept;
    [[nodiscard]] std::expected<std::vector<std::filesystem::path>, Failure> declare(const std::filesystem::path& dir) const;
    [[nodiscard]] const Manifest& manifest() const noexcept;
    [[nodiscard]] abi::OmuiRuntime handle() const noexcept;
    struct Impl;
  };
}
