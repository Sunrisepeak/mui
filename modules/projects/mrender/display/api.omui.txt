namespace mrender {

    Path& add_rect(const Rect& r);

    Path& add_ellipse(const Rect& r);
    Path& add_circle(Point center, float radius);
    [[nodiscard]] Path transformed(const Affine& m) const;
    [[nodiscard]] Rect bounds() const noexcept;

    [[nodiscard]] std::span<const PathVerb> verbs() const noexcept;
    [[nodiscard]] std::span<const Point> points() const noexcept;
    void clear() noexcept;
  };
  struct StrokeStyle {
    float width = 1;
    LineCap cap = LineCap::butt;
    LineJoin join = LineJoin::miter;
    float miter_limit = 4;
    std::span<const float> dashes;
    float dash_offset = 0;
    float trim_start = 0;
    float trim_end = 1;
    float trim_offset = 0;
  };
  struct GradientStop {
    float offset = 0;
    Color color;
  };
  enum class GradientKind : std::uint8_t { linear, radial, conic };
  enum class Spread : std::uint8_t { pad, repeat, reflect };
  struct Gradient {
    GradientKind kind = GradientKind::linear;
    Point start;
    Point end;
    float radius = 0;
    Spread spread = Spread::pad;
    std::span<const GradientStop> stops;
  };
  struct ImageRef {
    std::uint32_t id = 0;
    friend constexpr bool operator==(ImageRef a, ImageRef b) noexcept;
  };
  struct FontRef {
    std::uint32_t id = 0;
    friend constexpr bool operator==(FontRef a, FontRef b) noexcept;
  };
  struct EffectRef {
    std::uint32_t id = 0;
  };
  enum class Sampling : std::uint8_t { nearest, linear };
  struct Material {
    ImageRef normal;
    Point light{…};
    float light_z = 0.5f;
    Color shadow{…};
    float shadow_strength = 0.6f;
    int bands = 2;
    float softness = 0.08f;
    Color rim{…};
    float rim_strength = 0.6f;
    float rim_width = 0.35f;
    Color key{…};
    friend constexpr bool operator==(const Material&, const Material&) noexcept = default;
  };
  struct WeightedPaint;
  inline constexpr std::size_t max_paint_mix_terms = 16;
  struct Paint {
    enum class Kind : std::uint8_t { solid, gradient, image, image_mask, mix };
    Kind kind = Kind::solid;
    Color color;
    Gradient gradient;
    ImageRef image;
    Affine image_transform;
    Sampling sampling = Sampling::linear;
    const Material* material = nullptr;
    mlibs::memory::View<const WeightedPaint> mix;
    float mix_opacity = 1;
    [[nodiscard]] static Paint solid(Color c) noexcept;
  };
  struct WeightedPaint {
    Paint paint;
    float weight = 0;
  };
  enum class PaintMixError : std::uint8_t { invalid_weight, not_normalized, nested_mix, term_budget, invalid_progress, output_too_small, unprepared, invalid_options, invalid_leaf };
  [[nodiscard]] std::expected<void, PaintMixError> validate_paint_mix(const Paint& paint) noexcept;
  enum class BlendMode : std::uint8_t { normal, multiply, screen, overlay, darken, lighten, color_dodge, color_burn, hard_light, soft_light, difference, exclusion, plus, erase, };
  struct Glyph {
    std::uint32_t id = 0;
    Point position;
    std::uint32_t cluster = 0, grapheme = 0, word = 0, line = 0;
  };
  struct GlyphStyle {
    Affine transform;
    float opacity = 1;
    Color tint{…};
    [[nodiscard]] bool plain() const noexcept;
    [[nodiscard]] bool translation_only() const noexcept;
  };
  struct Ring {
    Rect outer;
    CornerRadii outer_radii;

    bool hollow = false;
  };

  enum class PixelAlign : std::uint8_t { none, device };

  };

  struct StrokeRoundRect {

  };

  };
