== omui.semantic:query
namespace omui::semantic {
  enum class QueryError : std::uint8_t { not_an_object, unknown_field, unknown_role, unknown_state, not_a_list, bad_rect, bad_point, nothing_to_wait_for };
  [[nodiscard]] constexpr mlibs::error::Description describe(QueryError error) noexcept;
  using QueryFailure = mlibs::error::Error<QueryError>;
  struct Query {
    std::string id;
    std::string under;
    std::optional<Role> role;
    std::string name;
    std::vector<State> states;
    std::vector<State> without;
    std::optional<Value> value;
    std::optional<Rect> within;
    std::optional<Point> at;
    bool interactive = false;
    std::size_t limit = 0;
    [[nodiscard]] bool matches(const spec::SemanticNode& node) const;
    [[nodiscard]] static std::expected<Query, QueryFailure> parse(const Value& v);
    [[nodiscard]] Value to_value() const;
  };
  struct Condition {
    std::optional<Query> query;
    bool absent = false;
    bool settled = false;
    [[nodiscard]] static std::expected<Condition, QueryFailure> parse(const Value& v);
  };
  [[nodiscard]] bool glob(std::string_view pattern, std::string_view text) noexcept;
}

== omui.semantic:tree
namespace omui::semantic {
  struct TreeOptions {
    bool compact = true;
  };
  struct SnapshotOptions {
    std::string root;
    int depth = -1;
    bool interactive = false;
    std::size_t offset = 0;
    std::size_t limit = 0;
    std::size_t children_limit = 0;
  };
  struct Snapshot {
    std::uint64_t revision = 0;
    std::vector<spec::SemanticNode> nodes;
    std::size_t total = 0;
    std::unordered_map<std::string, std::size_t> children_total;
    [[nodiscard]] Value to_value() const;
  };
  struct Step {
    std::string id;
    std::string action;
    Value args;
    std::string answer;
  };
  struct WaitState {
    bool satisfied = false;
    std::string why;
  };
  [[nodiscard]] Value changes_between(const Snapshot& before, const Snapshot& after);
  class Tree {
    explicit Tree(Runtime& runtime, TreeOptions options = {…});
    [[nodiscard]] Snapshot snapshot(const SnapshotOptions& options = {…});
    [[nodiscard]] spec::SemanticDelta changes();
    [[nodiscard]] std::vector<spec::SemanticNode> find(const Query& query);
    [[nodiscard]] std::optional<spec::SemanticNode> node(std::string_view id);
    Envelope act(std::string_view id, std::string_view action, const Invocation& call = {…});
    Envelope transaction(std::span<const Step> steps, const Invocation& call = {…}, std::string label = {…}, std::span<const Condition> expect = {…});
    Envelope sequence(std::span<const Step> steps, const Invocation& call = {…}, std::span<const Condition> expect = {…});
    Envelope preview(std::span<const Step> steps, const Invocation& call = {…}, const std::function<Value()>& look = {…});
    [[nodiscard]] Value state();
    Value restore(const Value& state, const Invocation& call = {…});
    [[nodiscard]] WaitState check(const Condition& condition);
    [[nodiscard]] std::vector<spec::Violation> violations();
    [[nodiscard]] std::uint64_t revision();
    [[nodiscard]] std::uint64_t digest();
    [[nodiscard]] Runtime& runtime() noexcept;
  };
}
