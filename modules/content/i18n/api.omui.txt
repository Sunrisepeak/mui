namespace mlibs::i18n {

    std::string decimal = ".", group = ",", minus = "-", plus = "+", percent = "%", exponential = "E", infinity = "∞", nan = "NaN";
    std::string decimal_pattern = "#,##0.###";
    std::string percent_pattern = "#,##0%";
    int minimum_grouping = 1;
    struct Compact {
      int magnitude = 0;
      Plural count = Plural::other;
      std::string pattern;
    };

  };
  struct CalendarData {
    std::array<std::array<std::string, 12>, 3> months, months_standalone;
    std::array<std::array<std::string, 7>, 3> days, days_standalone;
    std::array<std::string, 7> days_short;
    std::array<std::string, 3> am, pm;
    std::array<std::string, 2> eras;
    std::array<std::string, 4> date_formats, time_formats, datetime_formats;
    std::vector<std::pair<std::string, std::string>> skeletons;
    struct Period {
      std::string name;
      int from = 0;
      int before = 0;
      bool at = false;
    };
    std::vector<Period> period_rules;
    std::vector<std::pair<std::string, std::string>> period_names;
    std::string gmt_format = "GMT{0}", gmt_zero = "GMT", hour_format = "+HH:mm;-HH:mm";
  };
  enum class TimeUnit : std::uint8_t { year, quarter, month, week, day, hour, minute, second };
  struct RelativeData {
    struct Unit {
      std::vector<std::pair<int, std::string>> named;

      std::vector<std::pair<Plural, std::string>> past;
    };
    std::array<std::array<Unit, 3>, 8> units;
  };
  enum class ListType : std::uint8_t { conjunction, disjunction, unit };
  struct ListData {
    struct Patterns {
      std::string start = "{0}, {1}", middle = "{0}, {1}", end = "{0}, {1}", two = "{0}, {1}";
    };
    std::array<std::array<Patterns, 3>, 3> types;
  };
  struct LocaleData {
    Locale locale;
    std::string cldr;

    NumberData number;
    CalendarData calendar;
    RelativeData relative;

== mlibs.i18n.api:message
namespace mlibs::i18n {
  using Arg = std::variant<std::monostate, bool, std::int64_t, double, std::string, CivilTime, std::vector<std::string>>;
  class Args {
    Args() = default;
    Args(std::initializer_list<std::pair<std::string_view, Arg>> values);
    Args& set(std::string_view name, Arg value);
    [[nodiscard]] const Arg* get(std::string_view name) const noexcept;
  };
  class Message {
    [[nodiscard]] static std::expected<Message, std::string> parse(std::string_view source);
    [[nodiscard]] std::string format(const Formatter& fmt, const Args& args = {…}, std::vector<std::string>* errors = nullptr) const;
    [[nodiscard]] const std::string& source() const noexcept;
    [[nodiscard]] std::vector<std::string> variables() const;
    struct Option {
      std::string name, value;
      bool variable = false;
    };
    struct Expression {
      enum class Kind : std::uint8_t { none, variable, literal } kind = Kind::none;
      std::string operand;
      std::string function;
      std::vector<Option> options;
    };
    struct Part {
      std::string text;
      std::optional<Expression> expression;
    };
    using Pattern = std::vector<Part>;
    struct Declaration {
      bool input = true;
      std::string name;
      Expression expression;
    };
    struct Variant {
      std::vector<std::optional<std::string>> keys;
      Pattern pattern;
    };
  };
}
