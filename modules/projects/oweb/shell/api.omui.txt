    std::string contents;
    std::vector<Link> sections;
    std::string turn;

    std::string document;

  };

  };

  inline constexpr std::string_view document_id = "main/page/main/doc";
  [[nodiscard]] std::expected<Parts, std::string> read(std::string_view projection, std::string_view document);
}
